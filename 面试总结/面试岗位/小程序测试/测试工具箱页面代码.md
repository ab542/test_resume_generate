## index.js

```javascript
// pages/testpage/index.js
Page({
  data: {
    sdkVersion: '',
    wechatVersion: '',
    platform: '',
    system: '',
    requestResults: [],
    setDataResult: '',
    storageResult: '',
    navResult: '',
    getLoading: false,
    timeoutLoading: false,
    checklist: [
      { id: 1, label: '首页加载正常，无白屏', done: false },
      { id: 2, label: '消息页红点显示正确', done: false },
      { id: 3, label: '我的页面登录态正常', done: false },
      { id: 4, label: '各页面下拉刷新正常', done: false },
      { id: 5, label: '切后台再回来状态保持', done: false },
      { id: 6, label: '弱网下兜底页展示正常', done: false }
    ],
    doneCount: 0
  },

  onLoad() {
    const info = wx.getSystemInfoSync()
    this.setData({
      sdkVersion: info.SDKVersion,
      wechatVersion: info.version,
      platform: info.platform,
      system: info.system.split(' ')[0]
    })
  },

  // ===== wx.request 测试 =====

  testLogin() {
    this.setData({ getLoading: true })
    const start = Date.now()
    wx.request({
      url: 'https://star.digiplus-intl.com/api/auth/login',
      method: 'POST',
      header: {
        'Content-Type': 'application/json'
      },
      data: {
        email: 'liying@ama-auto.com',
        password: '3149390154Li'
      },
      timeout: 10000,
      success: (res) => {
        const cost = Date.now() - start
        if (res.statusCode === 200 && res.data.code === 0) {
          // 登录成功，存 token
          const token = res.data.data.token || res.data.data.access_token
          if (token) {
            wx.setStorageSync('star-token', token)
          }
          this.addResult('登录', cost, true)
        } else {
          this.addResult('登录', cost, false)
        }
      },
      fail: (e) => this.addResult('登录', Date.now() - start, false),
      complete: () => this.setData({ getLoading: false })
    })
  },

  testTimeout() {
    this.setData({ timeoutLoading: true })
    const start = Date.now()
    wx.request({
      url: 'https://httpbin.org/delay/10',
      timeout: 3000,
      success: () => this.addResult('超时(预期失败)', Date.now() - start, false),
      fail: (e) => this.addResult('超时(预期失败)', Date.now() - start,
        e.errMsg.includes('timeout')),
      complete: () => this.setData({ timeoutLoading: false })
    })
  },

  test404() {
    const start = Date.now()
    wx.request({
      url: 'https://httpbin.org/status/404',
      success: (res) => this.addResult('404', Date.now() - start, res.statusCode === 404),
      fail: (e) => this.addResult('404', Date.now() - start, false)
    })
  },

  testConcurrent() {
    this.setData({ requestResults: [] })
    const count = 5
    const starts = {}
    for (let i = 0; i < count; i++) {
      const id = `并发#${i + 1}`
      starts[id] = Date.now()
      wx.request({
        url: 'https://httpbin.org/get',
        success: () => this.addResult(id, Date.now() - starts[id], true),
        fail: (e) => this.addResult(id, Date.now() - starts[id], false)
      })
    }
  },

  addResult(name, time, ok) {
    const results = [...this.data.requestResults, { name, time, ok }]
    this.setData({ requestResults: results })
  },

  // ===== setData 测试 =====

  testSetData(e) {
    const kb = parseInt(e.currentTarget.dataset.kb)
    const data = 'a'.repeat(kb * 1024)
    const start = Date.now()
    this.setData({ testData: data }, () => {
      const cost = Date.now() - start
      this.setData({ setDataResult: `${kb}KB → setData 回调耗时 ${cost}ms` })
    })
  },

  // ===== Storage 测试 =====

  testStorageWrite() {
    const start = Date.now()
    const data = Array.from({ length: 100 }, (_, i) => ({
      id: i, name: `商品${i}`, price: Math.random() * 100
    }))
    wx.setStorageSync('test_batch_data', data)
    this.setData({ storageResult: `写入 100 条，耗时 ${Date.now() - start}ms` })
  },

  testStorageRead() {
    const start = Date.now()
    const data = wx.getStorageSync('test_batch_data')
    this.setData({
      storageResult: `读取 ${data ? data.length : 0} 条，耗时 ${Date.now() - start}ms`
    })
  },

  testStorageInfo() {
    wx.getStorageInfo({
      success: (res) => {
        this.setData({
          storageResult: `已用 ${res.currentSize}KB / ${res.limitSize}KB`
        })
      }
    })
  },

  testStorageClear() {
    wx.clearStorageSync()
    this.setData({ storageResult: '已清空全部 Storage' })
  },

  // ===== 导航测试 =====

  testNavigateTo() {
    this.setData({ navResult: `当前页面栈深度: ${getCurrentPages().length}` })
    wx.navigateTo({
      url: '/pages/search/index',    // 跳分包页面——必须是 非tabBar 页面
      fail: (e) => this.setData({ navResult: `navigateTo 失败: ${e.errMsg}` })
    })
  },

  testRedirectTo() {
    wx.redirectTo({
      url: '/pages/search/index',    // redirectTo 也不能跳 tabBar 页面
      fail: (e) => this.setData({ navResult: `redirectTo 失败: ${e.errMsg}` })
    })
  },

  testSwitchTab() {
    wx.switchTab({
      url: '/pages/my/index',        // switchTab 只能跳 tabBar 页面
      fail: (e) => this.setData({ navResult: `switchTab 失败: ${e.errMsg}` })
    })
  },

  testDeepStack() {
    const pages = getCurrentPages()
    this.setData({ navResult: `当前栈深度: ${pages.length}，开始连续跳转...` })
    let count = 0
    const go = () => {
      if (count >= 9) {
        this.setData({ navResult: this.data.navResult + `\n已跳转 ${count} 次，已达上限` })
        return
      }
      count++
      wx.navigateTo({
        url: '/pages/search/index',   // 用非tabBar页面测页面栈上限
        success: () => setTimeout(go, 300),
        fail: (e) => {
          this.setData({ navResult: this.data.navResult + `\n第 ${count} 次跳转失败: ${e.errMsg}` })
        }
      })
    }
    go()
  },

  // ===== 组件测试 =====

  testToast() {
    wx.showToast({ title: '测试消息', icon: 'none', duration: 2000 })
  },

  testModal() {
    wx.showModal({
      title: '测试弹窗',
      content: '这是 modal 的内容',
      success: (res) => {
        wx.showToast({ title: res.confirm ? '点击了确定' : '点击了取消', icon: 'none' })
      }
    })
  },

  testLoading() {
    wx.showLoading({ title: '加载中...' })
    setTimeout(() => { wx.hideLoading() }, 3000)
  },

  // ===== Checklist =====

  toggleCheck(e) {
    const id = e.currentTarget.dataset.id
    const checklist = this.data.checklist.map(item =>
      item.id === id ? { ...item, done: !item.done } : item
    )
    this.setData({
      checklist,
      doneCount: checklist.filter(i => i.done).length
    })
  }
})
```

## 功能一览

| 模块 | 测试目标 |
|------|---------|
| 环境信息栏 | 快速查看当前基础库版本、微信版本、系统 |
| wx.request 测试 | 正常 GET、超时、404、并发 5 个请求 |
| setData 测试 | 传输 64KB ~ 1MB 的数据，观察耗时 |
| Storage 测试 | 读写 100 条数据、查看占用、清空 |
| 导航测试 | navigateTo、redirectTo、连续跳转 10 层 |
| 组件测试 | Toast、Modal、Loading |
| 回归 Checklist | 6 条手工检查项，可勾选，统计完成度 |
